export default ({ workflows }) => {

    const prompt = w.text.match(/prompt:\s*\|([\s\S]*?)(\n\s{2,}\w+:|$)/)?.[1] || "";
    const refs = (prompt.match(/[\w./-]+\.(md|mjs|js|json|yml)/g) || []).filter((v, i, a) => a.indexOf(v) === i);
    if (refs.length > 3) out.push(`${w.file} prompt references ${refs.length} files; two (STATE.md plus one scope) is the diet`);
  }
  return out;
};
