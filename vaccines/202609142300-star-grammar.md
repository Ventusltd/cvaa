Disease

Related data views drift into incompatible controls, relationship labels, random layouts or status colours. A selected permanent key can no longer move consistently between views, and displayed quantities may lose their link to a dated source record.

Symptom

Two views name the same relationship differently; a page calls `Math.random()` and changes on reload; red or green is used as a correctness judgement; or a Star folder has neither a publication/provenance record nor a README stating what was and was not measured.

Antibody

```js
export default function (ctx) {
  const findings = [];
  const paths = new Set(ctx.paths || []);
  const legend = ["contains", "depends on", "uses", "used by", "shared line"];
  const rootStar = [...paths].some(p => p === "star.json" || p.startsWith("stars/") || p.startsWith("sun/"));
  const rootPage = rootStar && ctx.files && typeof ctx.files.index === "string" ? ctx.files.index : "";
  if (/Math\.random\s*\(/.test(rootPage)) findings.push("index.html: Math.random() is not reproducible; use a seeded function and retain the seed in the URL");
  if (/\b(pass|fail|failed|broken|correct|incorrect)\b[^\n]{0,40}(#f00|#ff0000|red|#0f0|#00ff00|green)/i.test(rootPage)) findings.push("index.html: red or green encodes a correctness judgement; use neutral category styling");
  const legendText = (rootPage.match(/class=["']legend["'][\s\S]{0,600}/) || [""])[0];
  if (legendText && !legend.every(word => legendText.includes(word))) findings.push(`index.html: relationship legend differs from the shared terms (${legend.join(", ")})`);

  const folders = new Set();
  for (const p of paths) {
    const match = p.match(/^(testcode\/2026091422\d{2})(?:\/|$)/);
    if (match && /(?:^|\/)(?:lenses\/[^/]+\.js|index\.html|core\.js|quantum\.js|ui\.js)$/.test(p)) folders.add(match[1]);
  }
  if (rootStar) folders.add("");
  for (const folder of folders) {
    const prefix = folder ? `${folder}/` : "";
    const hasProvenance = paths.has(`${prefix}publication.json`) || paths.has(`${prefix}data/provenance.json`) || paths.has(`${prefix}provenance.json`) || (!folder && paths.has("sun/provenance.json"));
    if (!hasProvenance) findings.push(`${folder || "."}: no publication.json or provenance.json; public quantities require dated source and hash records`);
    if (!paths.has(`${prefix}README.md`)) findings.push(`${folder || "."}: no README.md stating what the Star measures and does not measure`);
  }
  return findings;
}
```
